import { createSlice } from "@reduxjs/toolkit";

interface AuthState {
  isAuthenticated: boolean;
}

const initialState: AuthState = {
  isAuthenticated: false
};

const authSlice = createSlice({
  name: "auth",
  initialState,
  reducers: {
    setAuthenticated(state, action: { payload: boolean }) {
      state.isAuthenticated = action.payload; // Use payload
    },
    clearAuthenticated(state) {
      state.isAuthenticated = false; // Simpler than returning initialState
    }
  }
});

export const { setAuthenticated, clearAuthenticated } = authSlice.actions;
export default authSlice.reducer;
